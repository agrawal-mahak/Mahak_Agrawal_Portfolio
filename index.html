// ERPDataGrid.jsx - High-Performance Enterprise Data Grid with Debounced Search & Memoization
import React, { useState, useMemo, useCallback } from 'react';
import { useERPState } from '../hooks/useERPState';
import { useDebounce } from '../hooks/useDebounce';

export const ERPDataGrid = ({ category = 'all' }) => {
    const { items, isLoading, page, totalRecords, fetchNextPage } = useERPState({ category });
    const [searchQuery, setSearchQuery] = useState('');

    // Debounce search query to prevent unnecessary re-renders on 10,000+ items
    const debouncedSearch = useDebounce(searchQuery, 300);

    const filteredItems = useMemo(() => {
        if (!debouncedSearch.trim()) return items;
        const term = debouncedSearch.toLowerCase();
        return items.filter((row) => 
            row.skuCode.toLowerCase().includes(term) || 
            row.title.toLowerCase().includes(term)
        );
    }, [items, debouncedSearch]);

    const handleSearchChange = useCallback((e) => {
        setSearchQuery(e.target.value);
    }, []);

    return (
        <div className="erp-grid-container">
            <header className="erp-grid-toolbar">
                <input 
                    type="search" 
                    className="erp-search-input" 
                    placeholder="Filter 10,000+ ERP inventory records..." 
                    value={searchQuery} 
                    onChange={handleSearchChange} 
                />
                <span className="record-counter">Total: {filteredItems.length} items</span>
            </header>

            {isLoading ? (
                <div className="grid-skeleton-loader">Loading ERP pipeline...</div>
            ) : (
                <table className="erp-table">
                    <thead>
                        <tr>
                            <th>SKU Code</th>
                            <th>Item Name</th>
                            <th>Stock Status</th>
                        </tr>
                    </thead>
                    <tbody>
                        {filteredItems.map((row) => (
                            <tr key={row.id}>
                                <td><code>{row.skuCode}</td></td>
                                <td>{row.title}</td>
                                <td><span className="stock-badge">{row.stockQuantity} in stock</span></td>
                            </tr>
                        ))}
                    </tbody>
                </table>
            )}
        </div>
    );
};
